1917class UnitTestSource
1918{
1919public:
1921 static UnitTestSource& GetInstance() { static UnitTestSource inst; return inst; }
1922
1923private:
1924 UnitTestSource()
1925 {
1926#if defined(_IUTEST_DEBUG)
1927 detail::iuDebugInitialize();
1928#else
1929 IUTEST_UNUSED_VAR(&detail::iuDebugInitialize);
1930 IUTEST_UNUSED_VAR(&detail::iuDebugBreakAlloc);
1931#endif
1932 TestEnv::event_listeners().set_default_result_printer(new DefaultResultPrintListener());
1933 }
1934
1935public:
1937 ~UnitTestSource()
1938 {
1939 TestEnv::event_listeners().set_default_result_printer(NULL);
1940 TestEnv::event_listeners().set_default_xml_generator(NULL);
1941 }
1942
1943public:
1947 void Initialize()
1948 {
1949 UnitTest::instance().Initialize();
1950 }
1951
1955 int Run()
1956 {
1957 SetUpDefaultListener();
1958 return UnitTest::instance().Run();
1959 }
1960
1961public:
1962 void SetUpDefaultXmlListener()
1963 {
1964 if( TestEnv::is_output_option_dirty() )
1965 {
1966 if( TestEnv::has_output_option() )
1967 {
1968 TestEnv::flush_output_option();
1969
1970 do
1971 {
1972#if defined(__WANDBOX__)
1973 if( StderrXmlGeneratorListener::SetUp() )
1974 {
1975 break;
1976 }
1977 if( StderrJunitXmlGeneratorListener::SetUp() )
1978 {
1979 break;
1980 }
1981#else
1982 if( DefaultXmlGeneratorListener::SetUp() )
1983 {
1984 break;
1985 }
1986 if( JunitXmlGeneratorListener::SetUp() )
1987 {
1988 break;
1989 }
1990#endif
1991 IUTEST_LOG_(WARNING) << "unrecognized output format \"" << TestEnv::get_output_option() << "\" ignored.";
1992 } while( detail::AlwaysFalse() );
1993 }
1994 else
1995 {
1996 TestEnv::event_listeners().Release(TestEnv::event_listeners().default_xml_generator());
1997 }
1998 }
1999 }
2000
2001 void SetUpBriefResultPrinter()
2002 {
2003 if( TestFlag::IsEnableFlag(TestFlag::BRIEF)
2004 && TestEnv::event_listeners().default_result_printer() != NULL )
2005 {
2006 TestEnv::event_listeners().set_default_result_printer(new BriefUnitTestResultPrinter());
2007 }
2008 }
2009
2010private:
2011 void SetUpDefaultListener()
2012 {
2013 SetUpDefaultXmlListener();
2014#if IUTEST_HAS_STREAM_RESULT
2015 StreamResultListener::SetUp();
2016#endif
2017 }
2018};
