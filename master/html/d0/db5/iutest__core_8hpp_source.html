396 typedef detail::RegisteredTestSuiteResolver<Fixture> Resolver;
397 const TestTypeId id = internal::GetTypeId<Fixture>();
398 UnitTest& unit_test = UnitTest::instance();
399 bool found = false;
400 for( int i = 0; i < unit_test.total_test_suite_count(); ++i )

402 const TestSuite* suite = unit_test.GetTestSuite(i);
403 if( detail::IsStringEqual(suite->name(), test_suite_name) )
404 {
405 TestSuite::FindOp matches = { id, test_suite_name };
406 if( !matches(suite) )

409 return NULL;
