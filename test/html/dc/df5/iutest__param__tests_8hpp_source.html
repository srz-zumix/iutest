315template<typename T>
316class ParamTestInstance : public IParamTestInfoData
317{
318 typedef T Tester;
319 typedef typename Tester::ParamType ParamType;
320 typedef detail::iuParamTestFactory<T> Factory;
321
322 // 各テストのインスタンス
323 class EachTest IUTEST_CXX_FINAL : public IParamTestInfoData::ParamEachTestBase<ParamType>
324 {
325 public:
326 EachTest(TestSuite* testsuite, const ::std::string& name, const char* file, int line)
327 : m_mediator(testsuite)
328 , m_info(&m_mediator, name, &m_factory, file, line)
329 {
330 UnitTest::instance().AddTestInfo(testsuite, &m_info);
331 }
332 private:
333 virtual void SetParam(const ParamType& param) IUTEST_CXX_OVERRIDE
334 {
335 m_factory.SetParam(param);
336 m_info.set_value_param(PrintToString(param).c_str());
337 }
338 private:
339 TestSuiteMediator m_mediator;
340 Factory m_factory;
341 TestInfo m_info;
342 };
343public:
344 ParamTestInstance() : IParamTestInfoData(""), m_file(NULL), m_line(0) {} // allow empty name
345 explicit ParamTestInstance(const char* testsuite_name, const char* file=NULL, int line=0)
346 : IParamTestInfoData(testsuite_name), m_file(file), m_line(line) {}
347
348private:
349 // TestSuite の作成
350 virtual TestSuite* MakeTestSuite(const ::std::string& testsuite_name, TestTypeId id
351 , SetUpMethod setup, TearDownMethod teardown) const IUTEST_CXX_OVERRIDE
352 {
353#if !defined(IUTEST_NO_EXPLICIT_FUNCTION_TEMPLATE_ARGUMENTS)
354 return UnitTest::instance().AddTestSuite<TestSuite>(testsuite_name, id, setup, teardown);
355#else
356 return UnitTest::instance().AddTestSuite(
357 testsuite_name, id, setup, teardown, detail::explicit_type<TestSuite>());
358#endif
359 }
360
361 // テストの作成登録
362 virtual IParamTestInfoData::EachTestBase* RegisterTest(TestSuite* testsuite
363 , const ::std::string& name) const IUTEST_CXX_OVERRIDE
364 {
365 EachTest* test = new EachTest(testsuite, name, m_file, m_line);
366 // new オブジェクトを管理してもらう
367 detail::iuPool::GetInstance().push(test);
368 return test;
369 }
370private:
371 const char* m_file;
372 int m_line;
373};
