692 static void ParseCommandLine(int* pargc, CharType** argv)
693 {
694 if( argv == NULL )
695 {
696 return;
697 }
698 int argc = *pargc;
699
700 for( int i=0; i < argc; )
701 {
702 if( ParseCommandLineElem(argv[i]) )
703 {
704 --argc;
705 // 見つかった場合、オプションを末尾に移動
706 for( int k=i; k < argc; ++k )
707 {
708 CharType* tmp = argv[k];
709 argv[k] = argv[k+1];
710 argv[k+1] = tmp;
711 }
712 }
713 else
714 {
715 ++i;
716 }
717 }
718 *pargc = argc;
719 }
