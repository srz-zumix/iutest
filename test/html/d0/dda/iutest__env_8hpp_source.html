759 static void ParseCommandLine(int* pargc, CharType** argv)
760 {
761 if( argv == NULL )
762 {
763 return;
764 }
765 int argc = *pargc;
766
767 for( int i=0; i < argc; )
768 {
769 if( ParseCommandLineElem(argv[i]) )
770 {
771 --argc;
772 // 見つかった場合、オプションを末尾に移動
773 for( int k=i; k < argc; ++k )
774 {
775 CharType* tmp = argv[k];
776 argv[k] = argv[k+1];
777 argv[k+1] = tmp;
778 }
779 }
780 else
781 {
782 ++i;
783 }
784 }
785 *pargc = argc;
786 }
